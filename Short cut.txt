16. Geolocation: 
What is navigator.geolocation?
It’s a built-in browser API provided by the navigator object that lets you access the user’s geographical location (with their permission).

17. <Link> ==> Don't render the whole page like <a> tag

<button onclick="show()">Get Location</button>
<p id="result"></p>

<script>
function show() {
  if (navigator.geolocation) {
    navigator.geolocation.getCurrentPosition(
      function(position) {
        document.getElementById("result").innerHTML =
          "Latitude: " + position.coords.latitude + "<br>" +
          "Longitude: " + position.coords.longitude;
      },
      function(error) {
        document.getElementById("result").innerHTML =
          "Error: " + error.message;
      }
    );
  } else {
    document.getElementById("result").innerHTML = "Geolocation not supported.";
  }
}
</script>
